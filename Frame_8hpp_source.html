115 // Allocates tightly packed host storage using the canonical plane layout
116 // for the requested format. Existing capacity is reused where possible.
117 // Invalid dimensions and arithmetic overflow are rejected without
118 // modifying the existing frame.
119 void resize(int width, int height, PixelFormat format = PixelFormat::BGR8) {
120 if (width <= 0 || height <= 0) {
121 throw std::invalid_argument("Frame dimensions must be positive");
122 }
123
124 std::array<PlaneLayout, MaxPlanes> nextPlanes{};
125 std::size_t nextPlaneCount = 0;
126 std::size_t storageSize = 0;
127
128 auto addPlane = [&](int planeWidth, int planeHeight, std::size_t bytesPerSample) {
129 const std::size_t stride =
130 checkedMultiply(static_cast<std::size_t>(planeWidth), bytesPerSample);
131 const std::size_t planeSize =
132 checkedMultiply(stride, static_cast<std::size_t>(planeHeight));
133 if (storageSize > std::numeric_limits<std::size_t>::max() - planeSize) {
134 throw std::length_error("Frame storage size overflow");
135 }
136 nextPlanes[nextPlaneCount++] = {storageSize, planeSize, stride, planeWidth,
137 planeHeight};
138 storageSize += planeSize;
139 };
140
141 switch (format) {
142 case PixelFormat::Gray8:
143 addPlane(width, height, 1);
144 break;
145 case PixelFormat::RGB8:
146 case PixelFormat::BGR8:
147 addPlane(width, height, 3);
148 break;
149 case PixelFormat::RGBA8:
150 case PixelFormat::BGRA8:
151 addPlane(width, height, 4);
152 break;
153 case PixelFormat::NV12: {
154 const int chromaWidth = width / 2 + width % 2;
155 const int chromaHeight = height / 2 + height % 2;
156 addPlane(width, height, 1);
157 addPlane(chromaWidth, chromaHeight, 2);
158 break;
159 }
160 case PixelFormat::YUV420P: {
161 const int chromaWidth = width / 2 + width % 2;
162 const int chromaHeight = height / 2 + height % 2;
163 addPlane(width, height, 1);
164 addPlane(chromaWidth, chromaHeight, 1);
165 addPlane(chromaWidth, chromaHeight, 1);
166 break;
167 }
168 default:
169 throw std::invalid_argument("Unsupported pixel format");
170 }
171
172 storage_.resize(storageSize);
173 width_ = width;
174 height_ = height;
175 format_ = format;
176 planes_ = nextPlanes;
177 planeCount_ = nextPlaneCount;
178 timestamp_.reset();
179 sequence_.reset();
180 }
