16// Bytes per pixel for the packed 8-bit layouts a writer accepts. Planar and
17// semi-planar layouts return zero: their plane strides are backend-specific and
18// the writers do not take them.
19[[nodiscard]] inline int packedBytesPerPixel(PixelFormat format) noexcept {
20 switch (format) {
21 case PixelFormat::Gray8:
22 return 1;
23 case PixelFormat::RGB8:
24 case PixelFormat::BGR8:
25 return 3;
26 case PixelFormat::RGBA8:
27 case PixelFormat::BGRA8:
28 return 4;
29 case PixelFormat::NV12:
30 case PixelFormat::YUV420P:
31 return 0;
32 }
33 return 0;
34}

36// Validates a frame against the configuration the writer was opened with and
37// reports the reason on the error stream, so every backend refuses the same
38// input with the same diagnostic.
39[[nodiscard]] inline bool validateFrame(const Frame& frame, const VideoWriterConfig& config,
40 const char* backend) {
41 if (frame.empty()) {
42 std::cerr << backend << ": refusing to write an empty frame" << std::endl;
43 return false;
44 }
45 if (packedBytesPerPixel(frame.format()) == 0) {
46 std::cerr << backend << ": planar pixel formats are not accepted by the writer"
47 << std::endl;
48 return false;
49 }
50 if (frame.width() != config.width || frame.height() != config.height) {
51 std::cerr << backend << ": frame is " << frame.width() << "x" << frame.height()
52 << " but the writer was opened for " << config.width << "x" << config.height
53 << std::endl;
54 return false;
55 }
56 return true;
57}

59// Lowercased file extension of a destination, including the leading dot, or an
60// empty string when it has none. Backends that pick a container from the file
61// name share this so they agree on what ".MP4" means.
62[[nodiscard]] inline std::string destinationExtension(const std::string& destination) {
63 const std::size_t dot = destination.find_last_of('.');
64 if (dot == std::string::npos) {
65 return {};
66 }
67 std::string extension = destination.substr(dot);
68 std::transform(extension.begin(), extension.end(), extension.begin(),
69 [](unsigned char letter) { return static_cast<char>(std::tolower(letter)); });
70 return extension;
71}
